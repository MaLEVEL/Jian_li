from langgraph.graph import StateGraph, END
from typing import TypedDict, Literal

class AgentState(TypedDict):
    jd_text: str
    resume_text: str
    jd_structured: dict      # 解析后的JD
    resume_structured: dict  # 解析后的简历
    match_score: float
    missing_skills: list
    advice: str
    intent: Literal["single", "batch", "advice_only"]

def build_graph():
    g = StateGraph(AgentState)
    g.add_node("router",      route_intent)
    g.add_node("parse_jd",    parse_jd)          # 调LLM，输出JSON
    g.add_node("parse_resume", parse_resume)
    g.add_node("match",       calculate_match)   # 调skill_kb工具对齐技能词
    g.add_node("advise",      generate_advice)   # RAG检索素材库+生成建议
    g.add_node("batch",       batch_scoring)

    g.set_entry_point("router")
    g.add_conditional_edges("router", lambda s: s["intent"], {
        "single": "parse_jd",
        "batch":  "batch",
        "advice_only": "advise",
    })
    g.add_edge("parse_jd", "parse_resume")
    g.add_edge("parse_resume", "match")
    g.add_edge("match", "advise")
    g.add_edge("advise", END)
    g.add_edge("batch", END)
    return g.compile()

def calculate_match(jd_skills: list, resume_skills: list) -> dict:
    # 先用同义词库对齐（"LangChain" ≈ "LangGraph"、"sklearn" ≈ "scikit-learn"）
    aligned_resume = expand_synonyms(resume_skills)
    matched, missing = [], []
    for skill, weight in jd_skills:
        (matched if skill in aligned_resume else missing).append((skill, weight))
    score = sum(w for _, w in matched) / sum(w for _, w in jd_skills) * 100
    return {"score": round(score, 1),
            "matched": matched, "missing": missing,
            "explain": f"命中{len(matched)}/{len(jd_skills)}项必会技能"}

from rank_bm25 import BM25Okapi
import jieba

class HybridRetriever:
    def __init__(self, vectorstore):
        self.vs = vectorstore

    def search(self, query: str, top_k: int = 10):
        # 1. BM25 关键词召回
        bm25_hits = self.bm25.get_scores(jieba.lcut(query))
        # 2. 向量语义召回
        vec_hits = self.vs.similarity_search_with_score(query, k=10)
        # 3. RRF 融合 + Rerank 截断到最终 top_k
        fused = rrf_fuse(bm25_hits, vec_hits)
        return self.reranker.rerank(query, fused)[:top_k]
